# dismoji Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Vencord userplugin adding an Emoji Kitchen mashup tab to Discord's expression picker — pick a left emoji, browse its ~475 real mashup images, click to insert into the message box.

**Architecture:** Four units with a hard boundary between pure data logic and Discord integration. `codec.ts` owns the binary format (encode + decode in one file, so it cannot drift). `kitchen.ts` owns lookups and URL construction — no React, no Discord imports, fully unit-tested. `MashupPicker.tsx` is a mount-agnostic React component. `index.tsx` mounts it two ways: an expression-picker webpack patch (primary) and a chat-bar button (fallback), because the patch is the only fragile part.

**Tech Stack:** TypeScript, React (Discord's), Vencord plugin API, Vitest, Node >=22, pnpm.

**Spec:** `docs/superpowers/specs/2026-07-25-dismoji-design.md`

## Global Constraints

- Vencord verified at commit `4b9c27d905d6255141617546227a56073916ebd4` (2026-07-20). Re-verify file:line references if working against a newer checkout.
- Node >=22, pnpm 11.9.0 (from Vencord `package.json:83-86`).
- `src/kitchen.ts` and `src/codec.ts` MUST NOT import React, Vencord, or Discord modules. They are the only testable units; keeping them pure is what makes them testable.
- URL rule: every codepoint *component* takes a `u` prefix — `codepoint.split("-").map(c => "u" + c).join("-")`. Never `u{whole-codepoint}` (breaks 24.1% of pairs).
- Pair orientation MUST be read from the stored flag bit, never inferred from index order (inferring breaks 57% of lookups).
- All settings and send paths operate on the URL string. Never fetch image bytes at send time.
- Plugin name in Vencord's plugin list: `EmojiMashup`. Repo/folder name: `dismoji`.

`src/codec.ts`:
```ts
export interface PairRecord {
    lo: number;
    hi: number;
    dateIndex: number;
    loIsUrlLeft: boolean;
}

export interface RawIndex {
    emoji: string[];
    names: string[];
    dates: string[];
    pairs: PairRecord[];
}

const MAGIC = 0x4a4d5344; // "DSMJ" little-endian
const VERSION = 1;

export function encodeIndex(raw: RawIndex): Uint8Array {
    const enc = new TextEncoder();
    const emojiBytes = raw.emoji.map(e => enc.encode(e));
    const nameBytes = raw.names.map(n => enc.encode(n));

    let size = 4 + 1 + 2 + 1 + 4;
    for (const b of emojiBytes) size += 1 + b.length;
    for (const b of nameBytes) size += 2 + b.length;
    size += raw.dates.length * 8;
    size += raw.pairs.length * 5;

    const buf = new ArrayBuffer(size);
    const view = new DataView(buf);
    const bytes = new Uint8Array(buf);
    let o = 0;

    view.setUint32(o, MAGIC, true); o += 4;
    view.setUint8(o, VERSION); o += 1;
    view.setUint16(o, raw.emoji.length, true); o += 2;
    view.setUint8(o, raw.dates.length); o += 1;
    view.setUint32(o, raw.pairs.length, true); o += 4;

    for (const b of emojiBytes) {
        view.setUint8(o, b.length); o += 1;
        bytes.set(b, o); o += b.length;
    }
    for (const b of nameBytes) {
        view.setUint16(o, b.length, true); o += 2;
        bytes.set(b, o); o += b.length;
    }
    for (const d of raw.dates) {
        bytes.set(enc.encode(d), o); o += 8;
    }
    for (const p of raw.pairs) {
        view.setUint16(o, p.lo, true); o += 2;
        view.setUint16(o, p.hi, true); o += 2;
        view.setUint8(o, (p.dateIndex & 0x7f) | (p.loIsUrlLeft ? 0 : 0x80)); o += 1;
    }
    return bytes;
}

export function decodeIndex(bytes: Uint8Array): RawIndex {
    const view = new DataView(bytes.buffer, bytes.byteOffset, bytes.byteLength);
    const dec = new TextDecoder();
    let o = 0;

    if (view.getUint32(o, true) !== MAGIC) throw new Error("dismoji: bad magic number");
    o += 4;
    const version = view.getUint8(o); o += 1;
    if (version !== VERSION) throw new Error(`dismoji: unsupported index version ${version}`);

    const emojiCount = view.getUint16(o, true); o += 2;
    const dateCount = view.getUint8(o); o += 1;
    const pairCount = view.getUint32(o, true); o += 4;

    const emoji: string[] = [];
    for (let i = 0; i < emojiCount; i++) {
        const len = view.getUint8(o); o += 1;
        emoji.push(dec.decode(bytes.subarray(o, o + len))); o += len;
    }
    const names: string[] = [];
    for (let i = 0; i < emojiCount; i++) {
        const len = view.getUint16(o, true); o += 2;
        names.push(dec.decode(bytes.subarray(o, o + len))); o += len;
    }
    const dates: string[] = [];
    for (let i = 0; i < dateCount; i++) {
        dates.push(dec.decode(bytes.subarray(o, o + 8))); o += 8;
    }
    const pairs: PairRecord[] = [];
    for (let i = 0; i < pairCount; i++) {
        const lo = view.getUint16(o, true); o += 2;
        const hi = view.getUint16(o, true); o += 2;
        const flags = view.getUint8(o); o += 1;
        pairs.push({ lo, hi, dateIndex: flags & 0x7f, loIsUrlLeft: (flags & 0x80) === 0 });
    }
    return { emoji, names, dates, pairs };
}
```

`src/kitchen.ts`:
```ts
import type { RawIndex } from "./codec";

export interface Mashup {
    partner: string;
    url: string;
    name: string;
}

export interface Kitchen {
    emoji: string[];
    nameOf(codepoint: string): string;
    categoryOf(codepoint: string): string;
    partnersOf(codepoint: string): Mashup[];
    urlFor(a: string, b: string): string | null;
    search(query: string): string[];
}

const BASE = "https://www.gstatic.com/android/keyboard/emojikitchen";

/** Every codepoint component takes its own `u` prefix. */
export function toUrlSegment(codepoint: string): string {
    return codepoint.split("-").map(c => "u" + c).join("-");
}

export function buildUrl(date: string, left: string, right: string): string {
    const l = toUrlSegment(left);
    return `${BASE}/${date}/${l}/${l}_${toUrlSegment(right)}.png`;
}

export function createKitchen(raw: RawIndex): Kitchen {
    const indexOf = new Map<string, number>();
    raw.emoji.forEach((e, i) => indexOf.set(e, i));

    // adjacency[i] = every pair touching emoji i
    const adjacency: number[][] = raw.emoji.map(() => []);
    raw.pairs.forEach((p, i) => {
        adjacency[p.lo].push(i);
        adjacency[p.hi].push(i);
    });

    const urlOf = (pi: number): string => {
        const p = raw.pairs[pi];
        const loCp = raw.emoji[p.lo];
        const hiCp = raw.emoji[p.hi];
        const date = raw.dates[p.dateIndex];
        return p.loIsUrlLeft ? buildUrl(date, loCp, hiCp) : buildUrl(date, hiCp, loCp);
    };

    const field = (i: number, n: number) => raw.names[i]?.split("\t")[n] ?? "";

    const findPair = (a: string, b: string): number | null => {
        const ia = indexOf.get(a);
        const ib = indexOf.get(b);
        if (ia === undefined || ib === undefined) return null;
        const lo = Math.min(ia, ib);
        const hi = Math.max(ia, ib);
        for (const pi of adjacency[lo]) {
            const p = raw.pairs[pi];
            if (p.lo === lo && p.hi === hi) return pi;
        }
        return null;
    };

    return {
        emoji: raw.emoji,

        nameOf: cp => field(indexOf.get(cp) ?? -1, 0),

        categoryOf: cp => field(indexOf.get(cp) ?? -1, 2),

        partnersOf(cp) {
            const i = indexOf.get(cp);
            if (i === undefined) return [];
            return adjacency[i].map(pi => {
                const p = raw.pairs[pi];
                const other = p.lo === i ? p.hi : p.lo;
                const partner = raw.emoji[other];
                return { partner, url: urlOf(pi), name: field(other, 0) };
            });
        },

        urlFor(a, b) {
            const pi = findPair(a, b);
            return pi === null ? null : urlOf(pi);
        },

        search(query) {
            const q = query.trim().toLowerCase();
            if (!q) return [];
            return raw.emoji.filter((_, i) => {
                const [alt, kw] = raw.names[i].split("\t");
                return alt.toLowerCase().includes(q)
                    || kw.toLowerCase().split(",").some(k => k.includes(q));
            });
        }
    };
}
```

            if (li === undefined || ri === undefined) continue;

            const lo = Math.min(li, ri);
            const hi = Math.max(li, ri);
            const key = `${lo}:${hi}`;
            if (seen.has(key)) continue;

            dateSet.add(c.date);
            seen.set(key, { lo, hi, dateIndex: -1, loIsUrlLeft: li === lo });
        }
    }
}

const dates = [...dateSet].sort();
const dateIndexOf = new Map(dates.map((d, i) => [d, i]));

// second pass to assign date indices now that the table is ordered
for (const entry of Object.values<any>(meta.data)) {
    for (const combos of Object.values<any>(entry.combinations ?? {})) {
        for (const c of combos) {
            if (!c.isLatest) continue;
            const li = indexOf.get(c.leftEmojiCodepoint);
            const ri = indexOf.get(c.rightEmojiCodepoint);
            if (li === undefined || ri === undefined) continue;
            const rec = seen.get(`${Math.min(li, ri)}:${Math.max(li, ri)}`)!;
            if (rec.dateIndex === -1) rec.dateIndex = dateIndexOf.get(c.date)!;
        }
    }
}

const pairs = [...seen.values()].sort((a, b) => a.lo - b.lo || a.hi - b.hi);

// Invariants: fail the build rather than ship a broken index.
if (emoji.length !== EXPECTED_EMOJI)
    throw new Error(`emoji count ${emoji.length}, expected ${EXPECTED_EMOJI} — upstream changed`);
if (pairs.length < EXPECTED_PAIRS_MIN)
    throw new Error(`only ${pairs.length} pairs, expected >= ${EXPECTED_PAIRS_MIN}`);
if (dates.length > EXPECTED_DATES_MAX)
    throw new Error(`${dates.length} dates exceeds the 7-bit dateIndex field`);
if (pairs.some(p => p.dateIndex < 0))
    throw new Error("some pairs never received a date index");

const raw: RawIndex = { emoji, names, dates, pairs };
const packed = encodeIndex(raw);
const gz = gzipSync(packed, { level: 9 });
const b64 = Buffer.from(gz).toString("base64");

writeFileSync("src/kitchenData.ts",
`// GENERATED by scripts/build-index.mts — do not edit.
// Source: ${SOURCE}
// Emoji Kitchen artwork (c) Google. Pair metadata from xsalazar/emoji-kitchen.
// emoji=${emoji.length} pairs=${pairs.length} dates=${dates.length}
export const INDEX_B64 = "${b64}";
`);

console.log(`emoji=${emoji.length} pairs=${pairs.length} dates=${dates.length}`);
console.log(`packed=${(packed.length / 1024).toFixed(1)} KB gzip=${(gz.length / 1024).toFixed(1)} KB b64=${(b64.length / 1024).toFixed(1)} KB`);
```

import { decodeIndex } from "./codec";
import { createKitchen, type Kitchen } from "./kitchen";
import { INDEX_B64 } from "./kitchenData";

let cached: Promise<Kitchen> | null = null;

/** Decompress + build the adjacency map once, on first tab open. */
export function loadKitchen(): Promise<Kitchen> {
    cached ??= (async () => {
        const raw = Uint8Array.from(atob(INDEX_B64), c => c.charCodeAt(0));
        const stream = new Blob([raw]).stream().pipeThrough(new DecompressionStream("gzip"));
        const bytes = new Uint8Array(await new Response(stream).arrayBuffer());
        return createKitchen(decodeIndex(bytes));
    })();
    return cached;
}
```

- [ ] **Step 1: Place the plugin inside Vencord**

**Do not use a symlink or junction, and do not name the dev tsconfig `tsconfig.json`.**
Both break the build. esbuild resolves Vencord's `@api/*`, `@utils/*`, `@webpack/*`
aliases from the nearest `tsconfig.json` found walking up from each source file
(they are declared at `Vencord/tsconfig.json:25`). A junction resolves to its real
path outside Vencord's tree, and a `tsconfig.json` beside `index.tsx` shadows
Vencord's — either way the build fails with `Could not resolve "@api/ChatButtons"`.
Both failure modes were reproduced against a real build.

The plugin sources therefore live at the repo root, and the repo is cloned directly
into `src/userplugins/` — the standard userplugin model. Vencord gitignores that
path (`.gitignore:22`) and its pnpm workspace globs only `packages/*`, so the repo's
own `package.json` and `node_modules` are inert there.

import * as DataStore from "@api/DataStore";

export interface Recent { left: string; right: string; url: string; }

const KEY = "dismoji-recents";
const LIMIT = 24;

export async function getRecents(): Promise<Recent[]> {
    return (await DataStore.get<Recent[]>(KEY)) ?? [];

export async function pushRecent(r: Recent): Promise<void> {
    const list = (await getRecents()).filter(x => x.url !== r.url);
    list.unshift(r);
    await DataStore.set(KEY, list.slice(0, LIMIT));
}
